{{- /* 分类索引页（/categories/）——书架。 2026-09-24 改版：分类即书、tag 即章节。每本书（分类）下面列它的章节， 章节由「该书文章的 tags 与目录第二层名求交集」得出（章节在磁盘上就是 content/post/{书}/{NN-章节}/ 这一层），自由标签不再出现在书架页。 - 章节按磁盘目录的 NN- 序号排序（即 CHAPTERS 里定的章序），不是按文章数； - 数字是「该章的文章数」；平铺书（无章节层）显示「N 篇 · 平铺」； - 书排序沿用旧规则：近 $windowDays 天更新篇数降序 → 最近更新日 → 标题。 ⚠️ hugo sort 只吃一对 KEY/ORDER，多级排序靠「稳定排序连续排三趟」， 从次要键到主要键，顺序不能倒。 */ -}} {{ define "main" }} {{- $cats := .Site.Taxonomies.categories -}} {{- $windowDays := 180 -}} {{- $cutoff := now.AddDate 0 0 (int (mul -1 $windowDays)) -}}

{{ .Title }}

{{ T "list.page" (len $cats) }}

{{- $sections := slice -}} {{- range $cats -}} {{- $recent := 0 -}} {{- $newest := 0 -}} {{- range .Pages -}} {{- $ts := .Lastmod.Unix -}} {{- if gt $ts $newest }}{{ $newest = $ts }}{{ end -}} {{- if ge $ts $cutoff.Unix }}{{ $recent = add $recent 1 }}{{ end -}} {{- end -}} {{- $sections = $sections | append (dict "term" . "title" .Page.Title "recent" $recent "newest" $newest) -}} {{- end -}} {{- $sections = sort $sections "title" "asc" -}} {{- $sections = sort $sections "newest" "desc" -}} {{- $sections = sort $sections "recent" "desc" -}}
{{ range $sections }} {{- $term := .term -}} {{- $recent := .recent -}} {{- $book := .title -}} {{- /* 该书在磁盘上的章节集合：content/post/{书}/ 下带 NN- 前缀的目录。 readDir 的根是项目根（不是 content/），所以路径要带 content/ 前缀。 顺便记下 NN- 序号，排序时按章序而不是字典序。 */ -}} {{- $bookDir := printf "content/post/%s" $book -}} {{- $chapterSet := dict -}} {{- range readDir $bookDir -}} {{- if .IsDir -}} {{- if .Name | findRE `^\d+-` -}} {{- $ch := .Name | replaceRE `^\d+-` "" -}} {{- $ord := .Name | replaceRE `^(\d+)-.*` "$1" -}} {{- $chapterSet = merge $chapterSet (dict $ch $ord) -}} {{- end -}} {{- end -}} {{- end -}} {{- /* 文章 tag 与章节求交集，统计每章篇数；GetTerms 而不是 .Params.tags， 后者是原始字符串，对不上 Hugo 归一化后的词条 */ -}} {{- $acc := dict -}} {{- range $term.Pages -}} {{- range .GetTerms "tags" -}} {{- $ch := .Page.Title -}} {{- if index $chapterSet $ch -}} {{- $n := 1 -}} {{- with index $acc $ch -}} {{- $n = add .count 1 -}} {{- end -}} {{- $acc = merge $acc (dict $ch (dict "title" $ch "count" $n "href" .Page.RelPermalink "ord" (index $chapterSet $ch))) -}} {{- end -}} {{- end -}} {{- end -}} {{- $rows := slice -}} {{- range $ch, $t := $acc -}} {{- $rows = $rows | append $t -}} {{- end -}} {{- $rows = sort $rows "ord" "asc" -}}
{{ $book }} 近 {{ $windowDays }} 天 {{ $recent }} 篇 · 共 {{ len $term.Pages }} 篇{{ if $rows }} · {{ len $rows }} 章{{ else }} · 平铺{{ end }}
{{ with $rows }}
{{ range . }} {{ .title }} {{ .count }} {{ end }}
{{ end }}
